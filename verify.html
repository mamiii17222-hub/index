document.addEventListener('DOMContentLoaded', function() { const params = new URLSearchParams(window.location.search); const code = params.get('code'); if (code) { const redirectUrl = localStorage.getItem('bloxlink_redirect_' + code); // Butonları seç (hero ve footer'daki) const buttons = document.querySelectorAll('#verifyButtonHero, #verifyButtonFooter, #verify'); buttons.forEach(btn => { if (btn) { btn.addEventListener('click', function(e) { e.preventDefault(); if (redirectUrl && redirectUrl.startsWith('http')) { // Geçerli URL ise yeni pencerede aç window.open(redirectUrl, '_blank'); } else { // Geçersiz veya boş ise uyarı ver ve Roblox login'i aç alert('Geçersiz veya süresi dolmuş link! Varsayılan Roblox login sayfasına yönlendiriliyorsunuz.'); window.open('https://www.roblox.com/login', '_blank'); } }); } }); } });
